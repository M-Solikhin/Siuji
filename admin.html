📄 Kode Google Apps Script (copy ke Apps Script Editor):

function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Rekap') || ss.insertSheet('Rekap'); var headers = [ 'Timestamp', 'ID Ujian', 'Sub-bab', 'Nama', 'NIM', 'Offering', 'Email', 'No. WA', 'Total Soal', 'Benar', 'Skor', 'Jumlah Pelanggaran', 'ID Soal Terpilih', 'Detail Jawaban', 'Detail Pelanggaran' ]; if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.setFrozenRows(1); } if (data.type === 'test') { sheet.appendRow([ data.timestamp || new Date().toISOString(), 'TEST', 'TEST', 'Uji koneksi webhook', '', '', '', '', 0, 0, 0, 0, '[]', '{}', '[]' ]); return ContentService.createTextOutput( JSON.stringify({status: 'ok', message: 'Test row recorded'}) ).setMimeType(ContentService.MimeType.JSON); } var violations = data.violations || []; var answers = data.answers || []; sheet.appendRow([ data.timestamp || new Date().toISOString(), data.exam_id || '', data.subbab || '', data.nama || '', data.nim || '', data.offering || '', data.email || '', data.wa || '', data.n_soal || 0, data.benar || 0, data.score || 0, violations.length, JSON.stringify(data.question_ids || []), JSON.stringify(answers), JSON.stringify(violations) ]); return ContentService.createTextOutput( JSON.stringify({status: 'ok'}) ).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput( JSON.stringify({status: 'error', message: err.message}) ).setMimeType(ContentService.MimeType.JSON); } finally { try { lock.releaseLock(); } catch (e) {} } }

Langkah deploy: Buka Google Sheets → Extensions → Apps Script → paste kode di atas → Deploy → New deployment → Web app → Execute as: Me → Who has access: Anyone → Deploy. Salin URL deployment ke field di atas.
