📄 Kode Google Apps Script (copy ke Apps Script Editor):

function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // --- Sheet Rekap Utama --- var sheet = ss.getSheetByName('Rekap') || ss.insertSheet('Rekap'); if (sheet.getLastRow() === 0) { sheet.appendRow(['Timestamp','Sub-bab','Nama','NIM','Offering', 'Email','No.WA','Total Soal','Benar','Skor', 'Jml Pelanggaran','Detail Jawaban','Detail Pelanggaran']); } var violations = data.violations || []; sheet.appendRow([ data.timestamp, data.subbab, data.nama, data.nim, data.offering, data.email, data.wa, data.n_soal, data.benar, data.score, violations.length, JSON.stringify(data.answers), JSON.stringify(violations) ]); return ContentService .createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({status:'error',msg:err.message})) .setMimeType(ContentService.MimeType.JSON); } }

Langkah deploy: Buka Google Sheets → Extensions → Apps Script → paste kode di atas → Deploy → New deployment → Web app → Execute as: Me → Who has access: Anyone → Deploy. Salin URL deployment ke field di atas.
